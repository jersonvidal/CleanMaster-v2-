CREATE TABLE tb_empleado (

-- Crear tabla tb_cliente
DROP TABLE IF EXISTS tb_cliente;
CREATE TABLE tb_cliente (
    cod_cli INT PRIMARY KEY AUTO_INCREMENT,
    dni_cli INT,
    nom_cli VARCHAR(35),
    correo_cli VARCHAR(30),
    tele_cli VARCHAR(9),
    fecha_cli DATE,
    cod_tipo INT,
    cod_emp INT,
    FOREIGN KEY (cod_tipo) REFERENCES tb_tipo_servicio(cod_tipo),
    FOREIGN KEY (cod_emp) REFERENCES tb_empleado(cod_emp) -- Relación con empleado
);

-- Crear tabla tb_enlace
DROP TABLE IF EXISTS tb_enlace;
CREATE TABLE tb_enlace (
    idenlace INT NOT NULL AUTO_INCREMENT,
    descripcion VARCHAR(45) DEFAULT NULL,
    ruta VARCHAR(45) DEFAULT NULL,
    tipo VARCHAR(1) DEFAULT NULL,
    PRIMARY KEY (idenlace)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Crear tabla tb_rol
DROP TABLE IF EXISTS tb_rol;
CREATE TABLE tb_rol (
    idrol INT NOT NULL AUTO_INCREMENT,
    descripcion VARCHAR(45) DEFAULT NULL,
    PRIMARY KEY (idrol)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Crear tabla tb_rol_enlace
DROP TABLE IF EXISTS tb_rol_enlace;
CREATE TABLE tb_rol_enlace (
    idrol INT NOT NULL,
    idenlace INT NOT NULL,
    PRIMARY KEY (idrol, idenlace),
    FOREIGN KEY (idrol) REFERENCES tb_rol(idrol),
    FOREIGN KEY (idenlace) REFERENCES tb_enlace(idenlace)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- Crear tabla tb_usuario
DROP TABLE IF EXISTS tb_usuario;
CREATE TABLE tb_usuario (
    cod_usu INT NOT NULL AUTO_INCREMENT,
    login VARCHAR(45) DEFAULT NULL,
    password VARCHAR(45) DEFAULT NULL,
    idrol INT DEFAULT NULL,
    cod_emp INT DEFAULT NULL,
    PRIMARY KEY (cod_usu),
    FOREIGN KEY (idrol) REFERENCES tb_rol(idrol),
    FOREIGN KEY (cod_emp) REFERENCES tb_empleado(cod_emp)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;
